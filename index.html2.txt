    <script>
        function cavarPoco() {
            const btn = document.getElementById('btn-acao');
            const bloco = document.getElementById('progresso-bloco');
            const barra = document.getElementById('barra');
            const porcentagem = document.getElementById('porcentagem');
            const statusTexto = document.getElementById('status-texto');
            const concluidoTexto = document.getElementById('concluido');
            const avatar = document.querySelector('.avatar-placeholder');

            btn.disabled = true;
            btn.style.opacity = '0.5';
            btn.innerText = 'Trabalhando...';
            bloco.style.display = 'block';

            let progresso = 0;
            const intervalo = setInterval(() => {
                progresso += 5;
                barra.style.width = progresso + '%';
                porcentagem.innerText = progresso + '%';

                if (progresso === 30) {
                    statusTexto.innerText = "🧱 Reforçando as paredes do poço...";
                } else if (progresso === 65) {
                    statusTexto.innerText = "💧 Sinais de umidade detectados!";
                    avatar.innerText = "🤑"; 
                } else if (progresso >= 100) {
                    clearInterval(intervalo);
                    statusTexto.innerText = "🚀 Poço concluído com maestria!";
                    concluidoTexto.style.display = 'block';
                    btn.innerText = 'Trabalho Concluído';
                }
            }, 150);
        }
    </script>
